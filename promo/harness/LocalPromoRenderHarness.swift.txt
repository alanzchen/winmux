// Render-only XCTest harness for the product video. It's kept as .swift.txt so SwiftPM and the
// app's tests never compile it. To use it, copy it into Sources/AppBundleTests/ in a scratch
// checkout of this repository (never commit it there) and run, for example:
//   WINMUX_RENDER_DIR="$PWD/promo-renders" WINMUX_RENDER_SCALE=3 WINMUX_RENDER_WINDOW_SCALE=3 \
//     swift test --arch arm64 --filter LocalPromoRenderHarness
// It renders the REAL WinMux SwiftUI views (WorkspaceSidebarView in Tabs mode, the drag's display
// rails and the destination list) from NEUTRAL FIXTURE data. Each state also writes its drop-target
// frames (tab rows, pinned tiles, group headers, gaps) as JSON, which the film uses to slice the
// renders and move rows instead of cross-fading whole images. Copy the output into promo/public/ui/.
// WINMUX_RENDER_ONLY=<name,...> renders selected states; the 8x badge close-up uses
// WINMUX_RENDER_SCALE=8 WINMUX_RENDER_WINDOW_SCALE=8 WINMUX_RENDER_ONLY=B04-after-click.
@MainActor
final class LocalPromoRenderHarness: XCTestCase {
    override func tearDown() async throws {
        config = defaultConfig
        workspaceSidebarOrganizationStore = .init()
        try await super.tearDown()
    }

    private func tab(_ name: String, _ windows: [WorkspaceSidebarWindowViewModel], on scope: String,
                     project: WorkspaceProjectId? = nil, pinned: Bool = false, pinOrder: Int? = nil,
                     visible: Bool = false, focused: Bool = false) -> WorkspaceSidebarWorkspaceViewModel {
        var workspace = WorkspaceSidebarWorkspaceViewModel(name: name, projectId: project ?? launch, displayName: name,
            sidebarLabel: "", isGeneratedName: true, monitorScopeId: scope, monitorName: nil,
            isFocused: focused, isVisible: visible, items: windows.map { .init(kind: .window($0)) })
        workspace.appearance.isFavorite = pinned
        workspace.appearance.pinOrder = pinOrder
        workspace.apps = uniqueWorkspaceSidebarApps(windows.map { .init(name: $0.appName, bundleId: nil, bundlePath: $0.appBundlePath) })
        if pinned {
            let display = scope == scopeA ? nameA : scope == scopeB ? nameB : nameC
            workspace.knownDisplay = .init(monitorScopeId: scope, displayName: display)
        }
        return workspace
    }

    // MARK: rendering

    private final class TargetBox { var targets: [WorkspaceSidebarDropTargetFrame] = []; var surface: CGRect? }

    private var scale: CGFloat { CGFloat(Double(ProcessInfo.processInfo.environment["WINMUX_RENDER_SCALE"] ?? "3") ?? 3) }

    /// An offscreen window that reports a higher backing scale, so SwiftUI rasterizes text, symbols and
    /// icons at that scale (a window-less host always rasterizes them at the main screen's 2x).
    private final class ScaledWindow: NSWindow {
        var scale: CGFloat = 2
        override var backingScaleFactor: CGFloat { scale }
    }
    private var windowScale: CGFloat? { ProcessInfo.processInfo.environment["WINMUX_RENDER_WINDOW_SCALE"].flatMap { Double($0) }.map { CGFloat($0) } }
    private var windows: [NSWindow] = []

    private func host<V: View>(_ view: V, size: CGSize) -> NSHostingView<some View> {
        let host = NSHostingView(rootView: view.frame(width: size.width, height: size.height)
            .environment(\.colorScheme, .light)
            .transaction { $0.animation = nil })
        host.frame = CGRect(origin: .zero, size: size)
        host.appearance = NSAppearance(named: .aqua)
        if let windowScale {
            let window = ScaledWindow(contentRect: CGRect(x: -10000, y: -10000, width: size.width, height: size.height),
                styleMask: [.borderless], backing: .buffered, defer: false)
            window.scale = windowScale
            window.isReleasedWhenClosed = false
            window.contentView = host
            windows.append(window)
        }
        host.layoutSubtreeIfNeeded()
        for _ in 0 ..< 8 { RunLoop.main.run(until: Date(timeIntervalSinceNow: 0.1)); host.layoutSubtreeIfNeeded() }
        return host
    }

    private func writePNG(_ host: NSView, size: CGSize, to url: URL) throws {
        let rep = try XCTUnwrap(NSBitmapImageRep(bitmapDataPlanes: nil, pixelsWide: Int(size.width * scale),
            pixelsHigh: Int(size.height * scale), bitsPerSample: 8, samplesPerPixel: 4, hasAlpha: true, isPlanar: false,
            colorSpaceName: .deviceRGB, bytesPerRow: 0, bitsPerPixel: 0))
        rep.size = size
        host.cacheDisplay(in: host.bounds, to: rep)
        try XCTUnwrap(rep.representation(using: .png, properties: [:])).write(to: url)
    }

    private func describe(_ kind: WorkspaceSidebarDropTargetKind) -> [String: Any] {
        switch kind {
            case let .workspace(name): return ["type": "workspace", "name": name]
            case let .tabCollection(id, _): return ["type": "collection", "id": id]
            case .newWorkspace: return ["type": "newTab"]
            case let .monitor(id): return ["type": "monitor", "id": id]
            case let .tabGap(_, _, gap): return ["type": "gap", "workspace": gap.workspaceName, "after": gap.isAfter]
            case let .pinnedTabs(_, gap, _):
                var d: [String: Any] = ["type": "pinned"]
                if let gap { d["workspace"] = gap.workspaceName; d["after"] = gap.isAfter }
                return d
        }
    }

    /// WINMUX_RENDER_ONLY=name1,name2 renders just those states (e.g. a high-scale close-up).
    private func wanted(_ name: String) -> Bool {
        guard let only = ProcessInfo.processInfo.environment["WINMUX_RENDER_ONLY"], !only.isEmpty else { return true }
        return only.split(separator: ",").contains { $0 == name }
    }

    private func renderSidebar(_ name: String, _ w: World, scope: String, preview: WorkspaceSidebarDropPreviewViewModel? = nil,
                               badges: WorkspaceSidebarDockBadgeModel, dir: URL) throws {
        guard wanted(name) else { return }
        let box = TargetBox()
        let actions = WorkspaceSidebarActions(setDropTargets: { box.targets = $0 }, setSurfaceFrame: { box.surface = $0 })
        let music = AppleMusicNowPlayingModel(isMusicRunning: { false }, requestStatus: { _ in nil }, requestArtwork: { .failed })
        let view = WorkspaceSidebarView(snapshot: snapshot(w, scope: scope, preview: preview), actions: actions,
            reduceMotionOverride: true, reduceTransparencyOverride: true, dockBadgeModel: badges,
            browserTabsModel: BrowserTabsModel(snapshots: [:]), musicPlayerModel: music)
        let size = CGSize(width: width, height: height)
        let h = host(view, size: size)
        try writePNG(h, size: size, to: dir.appendingPathComponent("\(name).png"))
        let json: [String: Any] = [
            "name": name, "width": width, "height": height, "scale": scale,
            "surface": box.surface.map { ["x": $0.minX, "y": $0.minY, "w": $0.width, "h": $0.height] } as Any,
            "targets": box.targets.map { t -> [String: Any] in
                var d = describe(t.kind)
                d["x"] = t.frame.minX; d["y"] = t.frame.minY; d["w"] = t.frame.width; d["h"] = t.frame.height
                return d
            },
        ]
        let data = try JSONSerialization.data(withJSONObject: json, options: [.prettyPrinted, .sortedKeys])
        try data.write(to: dir.appendingPathComponent("\(name).json"))
    }

    private func renderRails(_ name: String, armingIndex: Int?, openIndex: Int?, dir: URL) throws {
        guard wanted(name) else { return }
        let hints = [
            WorkspaceSidebarDropDestinationHint(id: scopeB, name: nameB, direction: .right),
            WorkspaceSidebarDropDestinationHint(id: scopeC, name: nameC, direction: .right),
        ]
        let frames = [CGRect(x: 0, y: 0, width: 30, height: height), CGRect(x: 32, y: 0, width: 30, height: height)]
        let model = WorkspaceSidebarDropDestinationHintsModel()
        model.set(.init(hints: hints, frames: frames, armingId: armingIndex.map { hints[$0].id },
            openId: openIndex.map { hints[$0].id }, columnFits: true, isRow: false))
        let size = CGSize(width: 62, height: height)
        let h = host(WorkspaceSidebarDropDestinationHintsView(model: model), size: size)
        try writePNG(h, size: size, to: dir.appendingPathComponent("\(name).png"))
    }

    private func renderList(_ name: String, _ w: World, preview: WorkspaceSidebarDropPreviewViewModel?, dir: URL) throws {
        guard wanted(name) else { return }
        var projection = snapshot(w, scope: scopeB, preview: preview)
        projection.activeProjectId = launch
        let column = WorkspaceSidebarDropDestinationColumnModel()
        column.set(.init(hint: .init(id: scopeB, name: nameB, direction: .right), surface: .dropDestination(generation: 1),
            projection: projection, width: 280))
        let size = CGSize(width: 280, height: height)
        let h = host(WorkspaceSidebarDropDestinationView(model: column, scroll: WorkspaceSidebarDropDestinationScrollModel()) { _, _ in },
            size: size)
        try writePNG(h, size: size, to: dir.appendingPathComponent("\(name).png"))
    }

    private func badgeModel() async -> WorkspaceSidebarDockBadgeModel {
        var badges = WorkspaceSidebarDockBadgeSnapshot()
        badges.insert(url: URL(fileURLWithPath: messages), label: "3")
        let fixed = badges
        let model = WorkspaceSidebarDockBadgeModel(read: { fixed })
        model.setEnabled(true, showsAppBadges: true)
        for _ in 0 ..< 20 where model.snapshot.labelsByPath.isEmpty { try? await Task.sleep(for: .milliseconds(20)) }
        return model
    }

    // MARK: the film's states

    func testRenderPromoStates() async throws {
        guard let path = ProcessInfo.processInfo.environment["WINMUX_RENDER_DIR"] else { throw XCTSkip("render dir unset") }
        _ = NSApplication.shared
        let dir = URL(fileURLWithPath: path)
        try FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
        let badges = await badgeModel()

        // Act 2-3 on display A.
        var w = World()
        try renderSidebar("A01-design-f1", w, scope: scopeA, badges: badges, dir: dir)
        w.focusedWindowA = 21; try renderSidebar("A01-design-f2", w, scope: scopeA, badges: badges, dir: dir)
        w.focusedWindowA = 22; try renderSidebar("A01-design-f3", w, scope: scopeA, badges: badges, dir: dir)
        w.focusedWindowA = 20
        w.visibleA = "roadmap"; try renderSidebar("A02-roadmap", w, scope: scopeA, badges: badges, dir: dir)
        var wr = w; wr.projectA = writing; try renderSidebar("A03-writing", wr, scope: scopeA, badges: badges, dir: dir)
        w.groupCollapsed = true; try renderSidebar("A04-collapsed", w, scope: scopeA, badges: badges, dir: dir)
        // Dragging Roadmap onto the pinned tiles: the line between tiles marks the spot.
        var pinDrag = WorkspaceSidebarDropPreviewViewModel(sourceWindowId: 31, label: "Roadmap", appName: "Notes",
            targetWorkspaceName: nil, targetsNewWorkspace: false, isTabGroup: false, windowCount: 1)
        pinDrag.targetsPinned = true
        pinDrag.targetPinnedGap = .init(workspaceName: "pin-cal", isAfter: true)
        try renderSidebar("A04b-pin-drop-preview", w, scope: scopeA, preview: pinDrag, badges: badges, dir: dir)
        w.roadmapPinned = true; try renderSidebar("A05-pinned", w, scope: scopeA, badges: badges, dir: dir)

        // Act 4: the drag of Release notes (hidden on A) to Studio Display.
        let source = WorkspaceSidebarDropPreviewViewModel(sourceWindowId: 30, label: "Release notes", appName: "Safari",
            targetWorkspaceName: nil, targetsNewWorkspace: false, isTabGroup: false, windowCount: 1)
        try renderSidebar("A06-drag-source", w, scope: scopeA, preview: source, badges: badges, dir: dir)
        try renderSidebar("B01-base", w, scope: scopeB, badges: badges, dir: dir)
        try renderSidebar("C01-base", w, scope: scopeC, badges: badges, dir: dir)
        try renderRails("R01-idle", armingIndex: nil, openIndex: nil, dir: dir)
        try renderRails("R02-hover-b", armingIndex: 0, openIndex: nil, dir: dir)
        try renderRails("R03-open-b", armingIndex: nil, openIndex: 0, dir: dir)
        try renderList("L01-list-b", w, preview: nil, dir: dir)
        var gap = WorkspaceSidebarDropPreviewViewModel(sourceWindowId: 30, label: "Release notes", appName: "Safari",
            targetWorkspaceName: nil, targetsNewWorkspace: false, targetProjectId: launch, targetMonitorScopeId: scopeB,
            isTabGroup: false, windowCount: 1)
        gap.targetGap = .init(workspaceName: "docs", isAfter: true)
        try renderList("L02-list-b-gap", w, preview: gap, dir: dir)
        w.releaseOn = "B"
        try renderSidebar("A07-after-drop", w, scope: scopeA, badges: badges, dir: dir)
        try renderSidebar("B02-after-drop", w, scope: scopeB, badges: badges, dir: dir)
        var wv = w; wv.releaseVisibleOnB = true; wv.visibleB = "release"
        try renderSidebar("B02v-after-drop-visible", wv, scope: scopeB, badges: badges, dir: dir)

        // Shared pins.
        w.shared = true
        try renderSidebar("A08-shared", w, scope: scopeA, badges: badges, dir: dir)
        try renderSidebar("B03-shared", w, scope: scopeB, badges: badges, dir: dir)
        w.chatOn = "A"; w.visibleA = "chat"; w.visibleB = "docs"
        try renderSidebar("A09-after-click", w, scope: scopeA, badges: badges, dir: dir)
        try renderSidebar("B04-after-click", w, scope: scopeB, badges: badges, dir: dir)
        w.pinsReordered = true
        try renderSidebar("A10-reordered", w, scope: scopeA, badges: badges, dir: dir)
        try renderSidebar("B05-reordered", w, scope: scopeB, badges: badges, dir: dir)
    }
